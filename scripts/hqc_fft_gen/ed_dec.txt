definition qw :: "nat \<Rightarrow> (12, bool) seq" where "qw x = from_nat x"

definition OB :: "(512, w64) seq \<Rightarrow> nat \<Rightarrow> (8, bool) seq" where
  "OB fx x = nth_list (seq_to_list (FFTEnc.transpose8
     (map_seq (\<lambda>r. FFTEnc.byteOf r (take'`{3,3,bool} (coerce_seq_len (take'`{6,6,bool} (coerce_seq_len (qw x))))))
        (nth_list (map (\<lambda>x. list_to_seq (map (FFTEnc.linmap TY(64) FFTEnc.decMat) (list_len 8 x)) :: (8, w64) seq) (group_list 64 8 (seq_to_list fx)))
          (to_nat (drop'`{6,6,bool} (coerce_seq_len (qw x))))))))
     (to_nat (drop'`{3,3,bool} (coerce_seq_len (take'`{6,6,bool} (coerce_seq_len (qw x))))))"

lemma decode_list: "seq_to_list (FFTEnc.decode fx)
   = rev (map (list_to_seq :: bool list \<Rightarrow> w64) (group_list 512 64 (concat (map seq_to_list (rev (map (OB fx) [0..<4096]))))))"
  by (simp add: FFTEnc.decode_def Let_def FFTEnc.unflat2_def seq_to_list comp_def OB_def[abs_def] qw_def)

lemma decode_bit:
  fixes fx :: "(512, w64) seq"
  assumes p: "p < 32768"
  shows "bitsof (L (FFTEnc.decode fx)) p = bit (seq_to_word (OB fx (p div 8))) (p mod 8)"
proof -
  define xs where "xs = map (seq_to_list \<circ> OB fx) [0..<4096]"
  define BL where "BL = concat (rev xs)"
  have lx: "\<forall>x\<in>set xs. length x = 8" by (auto simp: xs_def)
  have lxs: "length xs = 4096" by (simp add: xs_def)
  have lBL: "length BL = 512 * 64"
  proof -
    have lc: "length (concat xs) = 4096 * 8" unfolding xs_def by (rule len_concat_upt) simp
    have "length BL = length (concat xs)" by (simp only: BL_def length_concat rev_map[symmetric] sum_list_rev)
    then show ?thesis using lc by simp
  qed
  define G where "G = group_list 512 64 BL"
  have sl: "seq_to_list (FFTEnc.decode fx) = rev (map (list_to_seq :: bool list \<Rightarrow> w64) G)"
    by (simp add: decode_list G_def BL_def xs_def rev_map comp_def)
  define i where "i = p div 64"
  define r where "r = p mod 64"
  have i: "i < 512" using p by (simp add: i_def)
  have r: "r < 64" by (simp add: r_def)
  have pir: "p = 64 * i + r" by (simp add: i_def r_def)
  have lg: "length (G ! (511 - i)) = 64" using i lBL by (simp add: G_def)
  have "L (FFTEnc.decode fx) ! i = seq_to_word (list_to_seq (G ! (511 - i)) :: w64)"
    using i by (simp add: L_def sl rev_nth G_def)
  then have "bitsof (L (FFTEnc.decode fx)) p = bit (seq_to_word (list_to_seq (G ! (511 - i)) :: w64)) r"
    using i by (simp add: bitsof_def L_def i_def r_def)
  also have "\<dots> = (G ! (511 - i)) ! (63 - r)"
    using r lg by (simp add: bit_s2w)
  also have "\<dots> = BL ! ((511 - i) * 64 + (63 - r))"
    using i r lBL by (simp add: G_def group_list_elem)
  also have "\<dots> = concat (rev xs) ! (8 * length xs - Suc p)"
  proof -
    have "(511 - i) * 64 + (63 - r) = 8 * length xs - Suc p" using i r pir lxs by simp
    then show ?thesis by (simp add: BL_def)
  qed
  also have "\<dots> = (xs ! (p div 8)) ! (8 - Suc (p mod 8))"
    using p lx lxs by (intro concat_rev_nth) simp_all
  also have "\<dots> = bit (seq_to_word (OB fx (p div 8))) (p mod 8)"
    using p by (simp add: xs_def bit_s2w)
  finally show ?thesis .
qed

lemma q6m: "q < 4096 \<Longrightarrow> to_nat (drop'`{6,6,bool} (coerce_seq_len (qw q))) = q mod 64"
  by (simp add: qw_def word_seq_convs seq_to_word.from_nat from_nat_word seq_to_word.drop' seq_to_word.coerce_seq_len
      unsigned_take_bit_eq take_bit_eq_mod unsigned_ucast_eq unat_mod unat_of_nat)

lemma q6d: "q < 4096 \<Longrightarrow> to_nat (take'`{6,6,bool} (coerce_seq_len (qw q))) = q div 64"
  by (simp add: qw_def word_seq_convs seq_to_word.from_nat from_nat_word seq_to_word.take' seq_to_word.coerce_seq_len
      unsigned_ucast_eq unat_drop_bit_eq drop_bit_eq_div unat_div take_bit_eq_mod unat_of_nat)

lemma rows_nth:
  fixes fx :: "(512, w64) seq"
  assumes i: "i < 64" and b: "b < 8"
  shows "nth_seq (nth_list (map (\<lambda>x. list_to_seq (map (FFTEnc.linmap TY(64) FFTEnc.decMat) (list_len 8 x)) :: (8, w64) seq)
                     (group_list 64 8 (seq_to_list fx))) i) b
       = FFTEnc.linmap TY(64) FFTEnc.decMat (nth_seq fx (8 * i + b))"
proof -
  have lg: "length (group_list 64 8 (seq_to_list fx) ! i) = 8" using i by simp
  show ?thesis using i b lg by (simp add: nth_seq.rep_eq group_list_elem mult.commute)
qed

lemma OB_bit:
  fixes fx :: "(512, w64) seq"
  assumes q: "q < 4096" and b: "b < 8"
  shows "bit (seq_to_word (OB fx q)) b
       = bit (seq_to_word (FFTEnc.linmap TY(64) FFTEnc.decMat (nth_seq fx (8 * (q mod 64) + b)))) (q div 64)"
proof -
  define T where "T = (take'`{6,6,bool} (coerce_seq_len (qw q)) :: (6, bool) seq)"
  define KB where "KB = (take'`{3,3,bool} (coerce_seq_len T) :: (3, bool) seq)"
  define RR where "RR = nth_list (map (\<lambda>x. list_to_seq (map (FFTEnc.linmap TY(64) FFTEnc.decMat) (list_len 8 x)) :: (8, w64) seq)
                     (group_list 64 8 (seq_to_list fx))) (q mod 64)"
  have tT: "to_nat T = q div 64" using q by (simp add: T_def q6d)
  have kb: "to_nat KB = q div 64 div 8" by (simp add: KB_def t3d tT)
  have jb: "to_nat (drop'`{3,3,bool} (coerce_seq_len T)) = q div 64 mod 8" by (simp add: t3m tT)
  have jl: "q div 64 mod 8 < 8" by simp
  have ob: "OB fx q = nth_seq (FFTEnc.transpose8 (map_seq (\<lambda>r. FFTEnc.byteOf r KB) RR)) (q div 64 mod 8)"
    using q by (simp add: OB_def nth_seq.rep_eq q6m jb RR_def KB_def T_def[symmetric])
  have "bit (seq_to_word (OB fx q)) b = bit (seq_to_word (nth_seq (map_seq (\<lambda>r. FFTEnc.byteOf r KB) RR) b)) (q div 64 mod 8)"
    using b jl by (simp add: ob transpose8_bit)
  also have "\<dots> = bit (seq_to_word (FFTEnc.byteOf (nth_seq RR b) KB)) (q div 64 mod 8)"
    using b by simp
  also have "\<dots> = bit (seq_to_word (nth_seq RR b)) (8 * (q div 64 div 8) + q div 64 mod 8)"
    using jl by (simp add: byteOf_bit kb)
  also have "\<dots> = bit (seq_to_word (FFTEnc.linmap TY(64) FFTEnc.decMat (nth_seq fx (8 * (q mod 64) + b)))) (q div 64)"
  proof -
    have "nth_seq RR b = FFTEnc.linmap TY(64) FFTEnc.decMat (nth_seq fx (8 * (q mod 64) + b))"
      unfolding RR_def by (rule rows_nth) (simp_all add: b)
    moreover have "8 * (q div 64 div 8) + q div 64 mod 8 = q div 64" by simp
    ultimately show ?thesis by simp
  qed
  finally show ?thesis .
qed

theorem decode_is_decodeIdx: "FFTEnc.decode fx = FFTEnc.decodeIdx fx"
proof (rule L_inj, rule bitsof_eqI)
  show "length (L (FFTEnc.decode fx)) = length (L (FFTEnc.decodeIdx fx))" by simp
next
  fix p assume "p < 64 * length (L (FFTEnc.decode fx))"
  then have p: "p < 32768" by simp
  have a: "8 * (p div 8 mod 64) + p mod 8 = p mod 512" using mod_mult2_eq[of p 8 64] by simp
  have b: "p div 8 div 64 = p div 512" using div_mult2_eq[of p 8 64] by simp
  show "bitsof (L (FFTEnc.decode fx)) p = bitsof (L (FFTEnc.decodeIdx fx)) p"
    using p by (simp add: decode_bit OB_bit decodeIdx_bit a b)
qed
